\section{Խնդրի դրվածքը}\label{sec:problem}

Այս բաժնում ձևակերպվում է աշխատանքի հիմնական խնդիրը։ Սկզբում քննարկվում է գծային դեպքը՝ Դիրիխլեի խնդիրը Պուասոնի հավասարման համար, և ապացուցվում են մաքսիմումի սկզբունքը ու կայունության գնահատականը։ Այնուհետև դիտարկվում է ոչ գծային խնդիրը, և ցույց է տրվում, որ գծային դեպքի արդյունքները թույլ են տալիս շատ կարճ ապացուցել համեմատման թեորեմը, լուծման եզակիությունը և կայունությունը նաև ոչ գծային դեպքում։

\subsection{Գծային դեպքը՝ Պուասոնի հավասարում}

Դիցուք $G=\{x=(x_1,x_2):\ 0<x_1<l_1,\ 0<x_2<l_2\}$ ուղղանկյունն է, $\Gamma$-ն՝ նրա եզրը, $\G=G\cup\Gamma$, իսկ $\Delta=\dfrac{\partial^2}{\partial x_1^2}+\dfrac{\partial^2}{\partial x_2^2}$՝ Լապլասի օպերատորը։ Դիրիխլեի խնդիրը Պուասոնի հավասարման համար հետևյալն է՝ գտնել $\G$-ում անընդհատ և $G$-ում երկու անգամ անընդհատ դիֆերենցելի $u(x)$ ֆունկցիա, որը բավարարում է

\begin{equation}\label{eq:poisson}
\Delta u=-f(x),\qquad x\in G,
\end{equation}
\begin{equation}\label{eq:poissonbc}
u(x)=\mu(x),\qquad x\in\Gamma,
\end{equation}
որտեղ $f\in C(\G)$ և $\mu\in C(\Gamma)$ տրված ֆունկցիաներ են։

Ստորև մեզ անհրաժեշտ կլինի ավելի ընդհանուր օպերատոր։ Դիցուք $c\in C(\G)$ և $c(x)\ge0$ ամբողջ $\G$-ում։ Նշանակենք
\begin{equation}\label{eq:Lc}
L_c w\equiv \Delta w-c(x)\,w.
\end{equation}
Երբ $c\equiv0$, ստանում ենք $L_0=\Delta$, և (\ref{eq:poisson}) հավասարումը գրվում է $L_0u=-f$ տեսքով։ Ոչ գծային խնդրի դեպքում $c$ գործակիցը կառաջանա գծայնացումից, ուստի կարևոր է, որ ստորև ստացվող գնահատականները կախված չլինեն $c$-ից։

Նշանակենք $R^2=\dfrac{l_1^2+l_2^2}{4}$ (ուղղանկյան կենտրոնից մինչև գագաթ հեռավորության քառակուսին) և դիտարկենք օգնական ֆունկցիան
\begin{equation}\label{eq:omega}
Q(x)=\frac14\Big[R^2-\Big(x_1-\frac{l_1}{2}\Big)^2-\Big(x_2-\frac{l_2}{2}\Big)^2\Big].
\end{equation}
Ուղղակի հաշվարկով $\Delta Q=-1$, իսկ $\G$-ի ցանկացած կետի հեռավորությունը կենտրոնից չի գերազանցում $R$-ը, ուստի
\begin{equation}\label{eq:C0}
0\le Q(x)\le\frac{R^2}{4}=C_0,\qquad C_0=\frac{l_1^2+l_2^2}{16},\qquad x\in\G.
\end{equation}

\begin{theorem}[մաքսիմումի սկզբունք]\label{thm:max}
Դիցուք $c\in C(\G)$, $c\ge0$, $w\in C^2(G)\cap C(\G)$ և $L_cw\ge0$ $G$-ում։ Այդ դեպքում
\begin{equation}\label{eq:max}
\max_{\G}w\ \le\ \max\Big\{0,\ \max_{\Gamma}w\Big\}.
\end{equation}
\end{theorem}

\begin{proof}
\emph{Քայլ 1 (խիստ անհավասարություն)։} Ենթադրենք $L_cw>0$ $G$-ում, և $\G$-ում $w$-ի մեծագույն արժեքը դրական է ու ընդունվում է $x^0\in G$ ներքին կետում։ Այդ կետում $\dfrac{\partial^2w}{\partial x_\alpha^2}\le0$ ($\alpha=1,2$), ուստի
\[L_cw(x^0)=\Delta w(x^0)-c(x^0)\,w(x^0)\ \le\ 0,\]
քանի որ $c\ge0$ և $w(x^0)>0$։ Սա հակասում է $L_cw>0$ պայմանին։ Հետևաբար, եթե $w$-ի մեծագույն արժեքը դրական է, ապա այն ընդունվում է $\Gamma$-ի վրա, այսինքն՝ (\ref{eq:max})-ը ճիշտ է։

\emph{Քայլ 2 (ընդհանուր դեպք)։} Նշանակենք $c_{\max}=\max_{\G}c$, $\gamma=\sqrt{c_{\max}+1}$ և $\varphi(x)=e^{\gamma x_1}$։ Այդ դեպքում
\[L_c\varphi=(\gamma^2-c)\,\varphi\ \ge\ (\gamma^2-c_{\max})\,\varphi=\varphi\ \ge\ 1.\]
Ցանկացած $\varepsilon>0$ համար $w_\varepsilon=w+\varepsilon\varphi$ ֆունկցիան բավարարում է $L_cw_\varepsilon\ge\varepsilon>0$, ուստի Քայլ 1-ից
\[\max_{\G}w\ \le\ \max_{\G}w_\varepsilon\ \le\ \max\Big\{0,\ \max_{\Gamma}w_\varepsilon\Big\}\ \le\ \max\Big\{0,\ \max_{\Gamma}w+\varepsilon e^{\gamma l_1}\Big\}.\]
Անցնելով $\varepsilon\to0$ սահմանին՝ ստանում ենք (\ref{eq:max})-ը։
\end{proof}

\begin{corollary}[նվազագույնի սկզբունք]\label{cor:min}
Եթե $c\ge0$, $w\in C^2(G)\cap C(\G)$ և $L_cw\le0$ $G$-ում, ապա $\min_{\G}w\ge\min\{0,\ \min_{\Gamma}w\}$։
\end{corollary}

\begin{proof}
Թեորեմ~\ref{thm:max}-ը կիրառում ենք $-w$ ֆունկցիայի նկատմամբ։
\end{proof}

\begin{theorem}[կայունություն]\label{thm:stab1}
Դիցուք $c\in C(\G)$, $c\ge0$, $g\in C(\G)$, իսկ $w\in C^2(G)\cap C(\G)$ ֆունկցիան բավարարում է $L_cw=-g$ հավասարմանը $G$-ում և $w=\nu$ պայմանին $\Gamma$-ի վրա։ Այդ դեպքում
\begin{equation}\label{eq:stab1}
\max_{\G}|w|\ \le\ \max_{\Gamma}|\nu|+C_0\max_{\G}|g|,\qquad C_0=\frac{l_1^2+l_2^2}{16}.
\end{equation}
\end{theorem}

\begin{proof}
Նշանակենք $m=\max_\Gamma|\nu|$, $M=\max_{\G}|g|$ և դիտարկենք $v_{\pm}=m+MQ\mp w$ ֆունկցիաները։ Քանի որ $\Delta Q=-1$, ապա
\[L_c(m+MQ)=-c\,m-M-c\,MQ\ \le\ -M,\]
որովհետև $c\ge0$, $m\ge0$, $Q\ge0$։ Բացի այդ, $L_c(\mp w)=\pm g$, ուստի $L_cv_\pm\le-M\pm g\le0$ $G$-ում։ $\Gamma$-ի վրա $v_\pm=m+MQ\mp\nu\ge m-|\nu|\ge0$։ Հետևանք~\ref{cor:min}-ից՝ $v_\pm\ge\min\{0,\min_\Gamma v_\pm\}=0$ ամբողջ $\G$-ում, այսինքն՝
\begin{equation*}
|w|\ \le\ m+MQ\ \le\ m+C_0M.
\end{equation*}
\end{proof}

\begin{remark}
(\ref{eq:stab1}) գնահատականի $C_0$ հաստատունը կախված չէ $c$-ից։ Միավոր քառակուսու համար ($l_1=l_2=1$) $C_0=1/8$, ինչը համընկնում է \cite[էջ~548]{bakh}-ում ստացված գնահատականի հետ։
\end{remark}

\begin{corollary}[եզակիություն]\label{cor:uniq1}
$L_cu=-f$, $u|_\Gamma=\mu$ խնդիրն ունի ամենաշատը մեկ լուծում։
\end{corollary}

\begin{proof}
Երկու լուծումների տարբերությունը բավարարում է Թեորեմ~\ref{thm:stab1}-ի պայմաններին $g=0$, $\nu=0$ դեպքում, ուստի (\ref{eq:stab1})-ից այն նույնաբար հավասար է զրոյի։
\end{proof}

\subsection{Ոչ գծային խնդիրը}

Դիտարկենք հետևյալ խնդիրը՝ գտնել $u\in C^2(G)\cap C(\G)$ ֆունկցիա, որը բավարարում է

\begin{equation}\label{eq:nl}
\Delta u-q(x,u)=-f(x),\qquad x\in G,
\end{equation}
\begin{equation}\label{eq:nlbc}
u(x)=\mu(x),\qquad x\in\Gamma.
\end{equation}
Այստեղ $f\in C(\G)$, $\mu\in C(\Gamma)$, իսկ $q(x,u)$-ն տրված ֆունկցիա է, որը ոչ գծային է ըստ $u$-ի։ Ենթադրում ենք, որ $q\in C(\G\times\mathbb R)$, $\dfrac{\partial q}{\partial u}$ ածանցյալը գոյություն ունի և անընդհատ է $\G\times\mathbb R$-ում, ընդ որում
\begin{equation}\label{eq:mono}
\dd{q}{u}(x,u)\ \ge\ 0\qquad\text{ցանկացած }x\in\G,\ u\in\mathbb R\text{ համար}.
\end{equation}

Հավասարման գլխավոր մասը Լապլասի օպերատորն է, ուստի այն էլիպսական է։ Ոչ գծայնությունը մտնում է միայն ստորին կարգի $q(x,u)$ անդամի մեջ, և այդ պատճառով հավասարումը կոչվում է \emph{կիսագծային} (semilinear)։ Երբ $q(x,u)=c(x)\,u$ և $c\ge0$, (\ref{eq:nl})-ը դառնում է $L_cu=-f$ գծային հավասարումը, իսկ (\ref{eq:mono})-ը՝ $c\ge0$ պայմանը։ Այսպիսով, (\ref{eq:mono})-ը գծային դեպքի $c\ge0$ պայմանի ոչ գծային անալոգն է, իսկ Պուասոնի հավասարումը համապատասխանում է $q\equiv0$ դեպքին։

\begin{example}[կիսահաղորդչային սարքի հավասարում]\label{ex:semi}
Կիսահաղորդչային սարքի մոդելավորման մեջ հավասարակշռային վիճակի $V_e$ էլեկտրական պոտենցիալը բավարարում է այսպես կոչված միջին դաշտի հավասարմանը \cite[գլ.~5, էջ~82--83]{mark}՝
\[\lambda^2\Delta V_e=\delta^2e^{V_e}-\delta^2e^{-V_e}-C(x),\]
որտեղ $\lambda,\delta>0$ սարքից կախված պարամետրեր են, իսկ $C(x)$-ը՝ դոնորային և ակցեպտորային խառնուրդների խտությունների տարբերությունն է (լեգիրման պրոֆիլը) \cite[էջ~73]{mark}։ Երկու մասերը բաժանելով $\lambda^2$-ի և նշանակելով $a=2\delta^2/\lambda^2$, $f=C/\lambda^2$, ստանում ենք (\ref{eq:nl}) տեսքը՝
\[\Delta u-a\sinh u=-f(x),\qquad u=V_e,\quad q(x,u)=a\sinh u.\]
Քանի որ $\dd{q}{u}=a\cosh u\ge a>0$, (\ref{eq:mono})-ը տեղի ունի։ Այս հավասարումը մեր հիմնական օրինակն է։ Իրական սարքում եզրի մի մասում պոտենցիալը տրվում է (Դիրիխլեի պայման), մյուս մասում՝ նրա նորմալ ածանցյալը \cite[էջ~83]{mark}։ Մենք դիտարկում ենք ամբողջ եզրի վրա Դիրիխլեի պայմանի պարզեցված դեպքը (\ref{eq:nlbc})։
\end{example}
